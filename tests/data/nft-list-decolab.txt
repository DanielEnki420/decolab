table inet decolab {
	set lan4 {
		type ipv4_addr
		flags interval
		elements = { 192.0.2.0/24 }
	}

	set lan6 {
		type ipv6_addr
		flags interval
		elements = { fc00::/7,
			     fe80::/10 }
	}

	chain input {
		type filter hook input priority filter - 1; policy drop;
		ct state established,related accept
		ct state invalid drop
		iifname "lo" accept
		iifname "tailscale0" accept
		iifname { "br-*", "docker0" } accept
		meta l4proto icmp accept
		meta l4proto ipv6-icmp accept
		udp dport 68 accept
		ip6 saddr fe80::/10 udp dport 546 accept
		udp dport 41641 accept
		ip saddr @lan4 tcp dport 22 accept
		ip6 saddr @lan6 tcp dport 22 accept
		ip saddr @lan4 udp dport 5353 accept
		ip6 saddr @lan6 udp dport 5353 accept
		ip saddr @lan4 meta l4proto { tcp, udp } th dport 53 accept
		ip6 saddr @lan6 meta l4proto { tcp, udp } th dport 53 accept
		ip saddr @lan4 tcp dport { 139, 445 } accept
		ip6 saddr @lan6 tcp dport { 139, 445 } accept
		ip saddr @lan4 tcp dport { 2283, 3001, 8053, 9000 } accept
		limit rate 4/minute burst 5 packets log prefix "decolab-drop in: "
	}

	chain forward {
		type filter hook forward priority filter - 1; policy accept;
		ct state established,related accept
		iifname { "br-*", "docker0", "tailscale0", "veth*" } accept
		ct status dnat ip saddr @lan4 meta l4proto { tcp, udp } ct original proto-dst 53 accept
		ct status dnat ip saddr @lan4 meta l4proto tcp ct original proto-dst { 139, 445 } accept
		ct status dnat ip saddr @lan4 meta l4proto tcp ct original proto-dst { 2283, 3001, 8053, 9000 } accept
		limit rate 4/minute burst 5 packets log prefix "decolab-drop fwd: "
		drop
	}
}
